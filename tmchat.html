TMChat Participant Security Rules

Database rules ensuring nobody can read or tamper with chats unless they are participant user1 or user2:

{
  "rules": {
    "users": {
      "$uid": {
        ".read": "auth != null",
        ".write": "$uid === auth.uid || root.child('users').child(auth.uid).child('role').val() === 'admin'"
      }
    },
    "chats": {
      "$chatId": {
        ".read": "auth != null && (data.child('user1').val() === auth.uid || data.child('user2').val() === auth.uid)",
        ".write": "auth != null && (data.child('user1').val() === auth.uid || data.child('user2').val() === auth.uid || !data.exists())",
        "messages": {
          "$msgId": {
            ".write": "auth != null && (newData.child('sender').val() === auth.uid || !newData.exists())"
          }
        }
      }
    }
  }
}
